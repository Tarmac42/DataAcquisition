cmake_minimum_required(VERSION 3.20)
project(DataAcquisition VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(acquisition_core src/config/SamplingConfig.cpp)
target_compile_features(acquisition_core PUBLIC cxx_std_20)
target_include_directories(acquisition_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(data-acquisition src/main.cpp)
target_link_libraries(data-acquisition PRIVATE acquisition_core)
if(BUILD_TESTING)
  add_executable(acquisition_contracts tests/contracts.cpp)
  target_link_libraries(acquisition_contracts PRIVATE acquisition_core)
  add_test(NAME acquisition_contracts COMMAND acquisition_contracts)
  add_test(NAME cli_help COMMAND data-acquisition --help)
  set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "No hardware drivers or background jobs")
  add_test(NAME cli_invalid_option COMMAND data-acquisition --unknown)
  set_tests_properties(cli_invalid_option PROPERTIES WILL_FAIL TRUE)
  foreach(header Error.hpp config/SamplingConfig.hpp devices/DeviceAdapter.hpp monitoring/AcquisitionMonitor.hpp services/AcquisitionService.hpp)
    string(MAKE_C_IDENTIFIER "${header}" name)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp "#include <tarmac/acquisition/${header}>\n")
    add_library(header_${name} OBJECT ${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp)
    target_link_libraries(header_${name} PRIVATE acquisition_core)
  endforeach()
endif()
